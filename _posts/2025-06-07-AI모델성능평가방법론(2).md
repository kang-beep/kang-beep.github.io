# 핵심 성능 지표의 완전 분석

1편에서 Confusion Matrix의 기본 개념을 살펴보았다면, 이번 편에서는 이를 바탕으로 한 구체적인 성능 지표들을 깊이 있게 분석해보겠습니다. 각 지표가 언제, 왜 중요한지 그리고 실제 비즈니스 상황에서 어떻게 활용해야 하는지 실전 예제와 함께 설명하겠습니다.

## 기본 성능 지표들의 수학적 정의와 의미

### 정확도 (Accuracy): 전체적인 성능의 척도

**수학적 정의:**
$$\text{Accuracy} = \frac{TP + TN}{TP + TN + FP + FN}$$

정확도는 전체 예측 중에서 맞춘 비율을 나타냅니다. 가장 직관적이고 이해하기 쉬운 지표이지만, 클래스 불균형이 있을 때는 매우 위험한 지표가 될 수 있습니다.

```python
# 2025-06-07: 정확도의 함정을 보여주는 예시
import numpy as np
from sklearn.metrics import accuracy_score, confusion_matrix

def demonstrate_accuracy_trap():
    """
    클래스 불균형 상황에서 정확도의 문제점 분석
    """
    # 신용카드 사기 탐지 시나리오 (사기: 1%, 정상: 99%)
    n_samples = 10000
    n_fraud = 100  # 실제 사기 거래
    n_normal = 9900  # 정상 거래
    
    # 실제 레이블
    y_true = [1] * n_fraud + [0] * n_normal
    
    # 세 가지 다른 모델의 예측
    models = {
        '게으른 모델': [0] * n_samples,  # 모든 거래를 정상으로 예측
        '조금 덜 게으른 모델': [1] * 20 + [0] * 80 + [1] * 200 + [0] * 9700,  # 사기 20개 탐지, 정상 200개 오분류
        '균형 잡힌 모델': [1] * 80 + [0] * 20 + [1] * 800 + [0] * 9100   # 사기 80개 탐지, 정상 800개 오분류
    }
    
    print("=== 신용카드 사기 탐지 시스템 비교 ===\n")
    
    for model_name, y_pred in models.items():
        accuracy = accuracy_score(y_true, y_pred)
        cm = confusion_matrix(y_true, y_pred)
        tn, fp, fn, tp = cm.ravel()
        
        # 비즈니스 임팩트 계산
        fraud_detection_rate = tp / (tp + fn) if (tp + fn) > 0 else 0
        false_alarm_rate = fp / (tn + fp) if (tn + fp) > 0 else 0
        
        # 비용 계산 (사기 1건당 평균 손실 $5000, 잘못된 경고 1건당 $10)
        fraud_loss = fn * 5000
        false_alarm_cost = fp * 10
        total_cost = fraud_loss + false_alarm_cost
        
        print(f"** {model_name} **")
        print(f"정확도: {accuracy:.4f} ({accuracy*100:.2f}%)")
        print(f"사기 탐지율: {fraud_detection_rate:.4f} ({fraud_detection_rate*100:.1f}%)")
        print(f"오경보율: {false_alarm_rate:.4f} ({false_alarm_rate*100:.2f}%)")
        print(f"예상 비용 손실: ${total_cost:,}")
        print(f"  - 놓친 사기로 인한 손실: ${fraud_loss:,}")
        print(f"  - 오경보로 인한 비용: ${false_alarm_cost:,}")
        print()

demonstrate_accuracy_trap()
```

**정확도 사용 시 주의사항:**
- 클래스가 균형적일 때만 신뢰할 수 있음
- 불균형 데이터에서는 다른 지표와 함께 사용 필수
- 단독 사용 시 심각한 오판 가능성

### 정밀도 (Precision): 양성 예측의 신뢰성

**수학적 정의:**
$$\text{Precision} = \frac{TP}{TP + FP}$$

정밀도는 "양성으로 예측한 것 중에서 실제로 양성인 비율"입니다. 거짓양성(False Positive)을 줄이는 것이 중요한 상황에서 핵심 지표입니다.

**정밀도가 중요한 상황:**
- 스팸 필터: 정상 메일을 스팸으로 분류하면 안 됨
- 의료진단: 건강한 사람을 환자로 오진하면 불필요한 치료
- 마케팅: 구매 의향이 없는 고객에게 광고하면 비용 낭비

### 재현율 (Recall/Sensitivity): 실제 양성의 탐지율

**수학적 정의:**
$$\text{Recall} = \frac{TP}{TP + FN}$$

재현율은 "실제 양성 중에서 모델이 찾아낸 비율"입니다. 거짓음성(False Negative)을 줄이는 것이 중요한 상황에서 핵심 지표입니다.

**재현율이 중요한 상황:**
- 암 검진: 환자를 놓치면 생명에 위험
- 보안 시스템: 침입자를 놓치면 큰 문제
- 사기 탐지: 사기를 놓치면 금전적 손실

```python
# 2025-06-07: 정밀도와 재현율의 트레이드오프 분석
import matplotlib.pyplot as plt
import numpy as np
from sklearn.metrics import precision_recall_curve, average_precision_score, confusion_matrix
from sklearn.linear_model import LogisticRegression
from sklearn.model_selection import train_test_split

def analyze_precision_recall_tradeoff():
    """
    정밀도와 재현율 간의 트레이드오프 관계 시각화
    """
    # 의료 진단 데이터 시뮬레이션
    np.random.seed(42)
    n_patients = 1000
    n_diseased = 200
    
    # 특성 생성 (증상 점수, 나이, 검사 결과 등)
    X_healthy = np.random.normal(0, 1, (n_patients - n_diseased, 3))
    X_diseased = np.random.normal(1.5, 1, (n_diseased, 3))
    
    X = np.vstack([X_healthy, X_diseased])
    y = np.array([0] * (n_patients - n_diseased) + [1] * n_diseased)
    
    # 모델 학습
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)
    model = LogisticRegression(random_state=42)
    model.fit(X_train, y_train)
    
    # 예측 확률 계산
    y_scores = model.predict_proba(X_test)[:, 1]
    
    # Precision-Recall 곡선 계산
    precisions, recalls, thresholds = precision_recall_curve(y_test, y_scores)
    ap_score = average_precision_score(y_test, y_scores)
    
    # 다양한 임계값에서의 성능 분석
    print("=== 진단 임계값별 성능 분석 ===\n")
    
    test_thresholds = [0.3, 0.5, 0.7, 0.9]
    for threshold in test_thresholds:
        y_pred = (y_scores >= threshold).astype(int)
        cm = confusion_matrix(y_test, y_pred)
        tn, fp, fn, tp = cm.ravel()
        
        precision = tp / (tp + fp) if (tp + fp) > 0 else 0
        recall = tp / (tp + fn) if (tp + fn) > 0 else 0
        f1 = 2 * (precision * recall) / (precision + recall) if (precision + recall) > 0 else 0
        
        print(f"임계값 {threshold:.1f}:")
        print(f"  정밀도: {precision:.3f} - 환자로 진단받은 사람 중 실제 환자 비율")
        print(f"  재현율: {recall:.3f} - 실제 환자 중 진단받은 비율")
        print(f"  F1-Score: {f1:.3f}")
        print(f"  놓친 환자: {fn}명, 오진: {fp}명")
        print()
    
    # 시각화
    plt.figure(figsize=(10, 6))
    plt.plot(recalls, precisions, linewidth=2, label=f'PR Curve (AP={ap_score:.3f})')
    plt.xlabel('재현율 (Recall)')
    plt.ylabel('정밀도 (Precision)')
    plt.title('의료 진단 시스템의 Precision-Recall 곡선')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.show()

analyze_precision_recall_tradeoff()
```

### 특이도 (Specificity): 음성 예측의 정확성

**수학적 정의:**
$$\text{Specificity} = \frac{TN}{TN + FP}$$

특이도는 "실제 음성 중에서 올바르게 음성으로 예측한 비율"입니다. 의료 분야에서 특히 중요하게 사용됩니다.

### F1-Score: 정밀도와 재현율의 조화평균

**수학적 정의:**
$$\text{F1-Score} = 2 \times \frac{\text{Precision} \times \text{Recall}}{\text{Precision} + \text{Recall}}$$

F1-Score는 정밀도와 재현율 사이의 균형점을 찾는 지표입니다. 두 지표가 모두 중요한 상황에서 사용합니다.

```python
# 2025-06-07: F1-Score와 변형들의 비교 분석
from sklearn.metrics import f1_score, fbeta_score, confusion_matrix

def analyze_f_scores():
    """
    F1-Score와 F-beta Score의 다양한 활용법
    """
    # 검색 엔진 관련성 평가 시나리오
    scenarios = {
        '검색 엔진 A (정밀도 중시)': {
            'y_true': [1]*100 + [0]*900,  # 관련 문서 100개, 무관 문서 900개
            'y_pred': [1]*80 + [0]*20 + [1]*50 + [0]*850,  # 관련 80개 검색, 무관 50개 검색
            'description': '정확한 결과만 보여주는 보수적 검색'
        },
        '검색 엔진 B (재현율 중시)': {
            'y_true': [1]*100 + [0]*900,
            'y_pred': [1]*95 + [0]*5 + [1]*200 + [0]*700,  # 관련 95개 검색, 무관 200개 검색
            'description': '관련 문서를 최대한 많이 찾는 포괄적 검색'
        },
        '검색 엔진 C (균형)': {
            'y_true': [1]*100 + [0]*900,
            'y_pred': [1]*87 + [0]*13 + [1]*120 + [0]*780,  # 관련 87개 검색, 무관 120개 검색
            'description': '정밀도와 재현율의 균형을 맞춘 검색'
        }
    }
    
    print("=== 검색 엔진 성능 비교 ===\n")
    
    for engine_name, scenario in scenarios.items():
        y_true, y_pred = scenario['y_true'], scenario['y_pred']
        
        # 기본 지표 계산
        cm = confusion_matrix(y_true, y_pred)
        tn, fp, fn, tp = cm.ravel()
        
        precision = tp / (tp + fp) if (tp + fp) > 0 else 0
        recall = tp / (tp + fn) if (tp + fn) > 0 else 0
        
        # 다양한 F-Score 계산
        f1 = f1_score(y_true, y_pred)
        f0_5 = fbeta_score(y_true, y_pred, beta=0.5)  # 정밀도 중시
        f2 = fbeta_score(y_true, y_pred, beta=2)      # 재현율 중시
        
        print(f"** {engine_name} **")
        print(f"설명: {scenario['description']}")
        print(f"정밀도: {precision:.3f}")
        print(f"재현율: {recall:.3f}")
        print(f"F1-Score: {f1:.3f} (균형)")
        print(f"F0.5-Score: {f0_5:.3f} (정밀도 중시)")
        print(f"F2-Score: {f2:.3f} (재현율 중시)")
        print()

analyze_f_scores()
```

## 고급 성능 지표들

### ROC 곡선과 AUC: 분류 성능의 종합적 평가

ROC(Receiver Operating Characteristic) 곡선은 다양한 임계값에서 TPR(True Positive Rate)과 FPR(False Positive Rate)의 관계를 보여줍니다.

- **TPR (재현율)**: TP / (TP + FN)
- **FPR**: FP / (FP + TN)
- **AUC**: ROC 곡선 아래 면적 (0.5-1.0, 높을수록 좋음)

```python
# 2025-06-07: ROC 곡선과 AUC 분석
import numpy as np
import matplotlib.pyplot as plt
from sklearn.metrics import roc_curve, auc, roc_auc_score

def analyze_roc_auc():
    """
    다양한 모델의 ROC 곡선과 AUC 비교
    """
    # 이진 분류 데이터 생성 (질병 진단)
    np.random.seed(42)
    n_healthy = 800
    n_sick = 200
    
    # 세 가지 다른 품질의 진단 시스템 시뮬레이션
    models = {
        '우수한 진단기': {
            'healthy_scores': np.random.normal(0.2, 0.1, n_healthy),  # 건강한 사람은 낮은 점수
            'sick_scores': np.random.normal(0.8, 0.1, n_sick)        # 환자는 높은 점수
        },
        '보통 진단기': {
            'healthy_scores': np.random.normal(0.3, 0.2, n_healthy),
            'sick_scores': np.random.normal(0.7, 0.2, n_sick)
        },
        '나쁜 진단기': {
            'healthy_scores': np.random.normal(0.45, 0.2, n_healthy),
            'sick_scores': np.random.normal(0.55, 0.2, n_sick)
        }
    }
    
    plt.figure(figsize=(12, 8))
    
    for model_name, scores in models.items():
        # 실제 레이블과 예측 점수 생성
        y_true = [0] * n_healthy + [1] * n_sick
        y_scores = np.concatenate([scores['healthy_scores'], scores['sick_scores']])
        
        # ROC 곡선 계산
        fpr, tpr, thresholds = roc_curve(y_true, y_scores)
        auc_score = auc(fpr, tpr)
        
        # 시각화
        plt.plot(fpr, tpr, linewidth=2, label=f'{model_name} (AUC={auc_score:.3f})')
        
        print(f"=== {model_name} 성능 분석 ===")
        print(f"AUC: {auc_score:.3f}")
        
        # 최적 임계값 찾기 (Youden's J statistic)
        optimal_idx = np.argmax(tpr - fpr)
        optimal_threshold = thresholds[optimal_idx]
        optimal_tpr = tpr[optimal_idx]
        optimal_fpr = fpr[optimal_idx]
        
        print(f"최적 임계값: {optimal_threshold:.3f}")
        print(f"최적점에서 TPR: {optimal_tpr:.3f}, FPR: {optimal_fpr:.3f}")
        print()
    
    # 대각선 (무작위 분류기) 추가
    plt.plot([0, 1], [0, 1], 'k--', linewidth=1, label='무작위 분류기 (AUC=0.5)')
    
    plt.xlabel('거짓양성률 (FPR)')
    plt.ylabel('참양성률 (TPR)')
    plt.title('다양한 진단 시스템의 ROC 곡선 비교')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.show()

analyze_roc_auc()
```

### Matthews Correlation Coefficient (MCC): 불균형 데이터의 진정한 성능

MCC는 클래스 불균형이 심한 상황에서도 신뢰할 수 있는 성능 지표입니다.

**수학적 정의:**
$$\text{MCC} = \frac{TP \times TN - FP \times FN}{\sqrt{(TP+FP)(TP+FN)(TN+FP)(TN+FN)}}$$

- 범위: -1 ~ +1
- +1: 완벽한 예측
- 0: 무작위 예측
- -1: 완전히 잘못된 예측

### 주요 카테고리별 상세 설명

#### 📊 지도학습 평가 특징
- **명확한 정답 존재**: 실제 레이블과 비교하여 정확한 성능 측정 가능
- **문제 유형별 특화**: 분류와 회귀 문제에 따라 다른 지표 사용
- **비즈니스 영향 고려**: FP와 FN의 비용이 다른 경우 가중치 적용

#### 🔍 비지도학습 평가 특징  
- **정답 부재**: 내재적(intrinsic) 지표와 외재적(extrinsic) 지표 활용
- **도메인 전문성 필요**: 결과의 해석에 도메인 지식 중요
- **여러 지표 조합**: 단일 지표로는 판단 어려워 다각도 평가 필수

#### 🎮 강화학습 평가 특징
- **환경과의 상호작용**: 에이전트의 행동과 환경 반응 모두 고려
- **장기적 관점**: 즉시 보상보다는 누적 보상이 중요
- **탐험-활용 균형**: 학습 과정에서의 탐험과 활용 간 균형 평가

#### 🌟 특수 모델 평가 특징
- **도메인 특화 지표**: 각 분야의 특성을 반영한 전용 지표 사용
- **인간 평가 포함**: 자동화된 지표와 함께 인간 평가자의 판단 활용
- **다차원 평가**: 정확성, 유창성, 관련성, 안전성 등 여러 차원 고려

### 이상 탐지 모델 평가

```python
# 2025-06-07: 이상 탐지 모델 성능 평가
import numpy as np
import matplotlib.pyplot as plt
from sklearn.ensemble import IsolationForest
from sklearn.metrics import precision_recall_curve, confusion_matrix

def evaluate_anomaly_detection():
    """
    이상 탐지 모델의 특수한 평가 방법들
    """
    # 네트워크 보안 데이터 시뮬레이션
    np.random.seed(42)
    
    # 정상 트래픽 (대부분)
    normal_data = np.random.multivariate_normal([0, 0], [[1, 0.1], [0.1, 1]], 900)
    
    # 이상 트래픽 (소수)
    anomaly_data = np.random.multivariate_normal([3, 3], [[0.5, 0], [0, 0.5]], 100)
    
    X = np.vstack([normal_data, anomaly_data])
    y_true = [0] * 900 + [1] * 100  # 0: 정상, 1: 이상
    
    # Isolation Forest 모델 학습
    iso_forest = IsolationForest(contamination=0.1, random_state=42)
    y_pred = iso_forest.fit_predict(X)
    anomaly_scores = iso_forest.score_samples(X)
    
    # 예측 결과 변환 (-1을 1로, 1을 0으로)
    y_pred_binary = [1 if pred == -1 else 0 for pred in y_pred]
    
    print("=== 이상 탐지 모델 평가 ===\n")
    
    # 기본 분류 성능
    cm = confusion_matrix(y_true, y_pred_binary)
    tn, fp, fn, tp = cm.ravel()
    
    precision = tp / (tp + fp) if (tp + fp) > 0 else 0
    recall = tp / (tp + fn) if (tp + fn) > 0 else 0
    f1 = 2 * (precision * recall) / (precision + recall) if (precision + recall) > 0 else 0
    
    print(f"기본 성능 지표:")
    print(f"정밀도: {precision:.3f} - 이상으로 분류된 것 중 실제 이상 비율")
    print(f"재현율: {recall:.3f} - 실제 이상 중 탐지된 비율")
    print(f"F1-Score: {f1:.3f}")
    print()
    
    # 이상 탐지 특화 지표들
    # 1. Precision at K (상위 K개 의심 케이스의 정밀도)
    k_values = [10, 20, 50]
    sorted_indices = np.argsort(-anomaly_scores)  # 이상 점수 높은 순으로 정렬
    
    print("Precision at K 분석:")
    for k in k_values:
        top_k_indices = sorted_indices[:k]
        top_k_true = [y_true[i] for i in top_k_indices]
        precision_at_k = sum(top_k_true) / k
        print(f"Precision@{k}: {precision_at_k:.3f} - 상위 {k}개 의심 케이스 중 실제 이상 비율")
    print()
    
    # 2. 이상 점수 분포 분석
    normal_scores = [anomaly_scores[i] for i in range(len(y_true)) if y_true[i] == 0]
    anomaly_scores_true = [anomaly_scores[i] for i in range(len(y_true)) if y_true[i] == 1]
    
    print("이상 점수 분포 분석:")
    print(f"정상 데이터 평균 이상 점수: {np.mean(normal_scores):.3f}")
    print(f"이상 데이터 평균 이상 점수: {np.mean(anomaly_scores_true):.3f}")
    print(f"점수 분리도: {np.mean(anomaly_scores_true) - np.mean(normal_scores):.3f}")

evaluate_anomaly_detection()
```
